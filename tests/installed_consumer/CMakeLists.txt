cmake_minimum_required(VERSION 3.20)
project(alligator_installed_consumer LANGUAGES CXX)
set(CMAKE_FIND_USE_PACKAGE_REGISTRY OFF)
set(CMAKE_FIND_USE_SYSTEM_PACKAGE_REGISTRY OFF)
set(CMAKE_FIND_USE_CMAKE_ENVIRONMENT_PATH OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
find_package(alligator CONFIG REQUIRED
    PATHS "${ALLIGATOR_RELOCATED_PREFIX}/${ALLIGATOR_PACKAGE_CMAKEDIR}" NO_DEFAULT_PATH)
add_executable(alligator_installed_consumer main.cpp)
target_link_libraries(alligator_installed_consumer PRIVATE alligator::alligator)
target_compile_definitions(alligator_installed_consumer PRIVATE NDEBUG)
set(feature_source "${CMAKE_CURRENT_BINARY_DIR}/backend_definitions.cpp")
file(WRITE "${feature_source}" "#include <buffet.hpp>\n")
string(REPLACE "|" ";" expected_definitions "${ALLIGATOR_EXPECTED_DEFINITIONS}")
foreach(backend IN ITEMS VULKAN METAL CUDA)
    set(expected_backend 0)
    set(expected_value 1)
    foreach(definition IN LISTS expected_definitions)
        if(definition MATCHES "^BUFFETALLIGATOR_HAS_${backend}(=|$)")
            set(expected_backend 1)
            if(definition MATCHES "^BUFFETALLIGATOR_HAS_${backend}=(.+)$")
                set(expected_value "${CMAKE_MATCH_1}")
            endif()
        endif()
    endforeach()
    file(APPEND "${feature_source}"
        "#if defined(BUFFETALLIGATOR_HAS_${backend}) != ${expected_backend}\n"
        "#error Installed package changed the ${backend} feature definition\n#endif\n")
    if(expected_backend)
        file(APPEND "${feature_source}"
            "#if BUFFETALLIGATOR_HAS_${backend} != (${expected_value})\n"
            "#error Installed package changed the ${backend} feature value\n#endif\n")
    endif()
endforeach()
target_sources(alligator_installed_consumer PRIVATE "${feature_source}")
set(installed_include "${ALLIGATOR_RELOCATED_PREFIX}/${ALLIGATOR_INSTALL_INCLUDEDIR}")
file(GLOB_RECURSE public_headers RELATIVE "${installed_include}"
    "${installed_include}/alligator/*.hpp" "${installed_include}/alligator/*.h")
list(APPEND public_headers alligator.hpp buffet.hpp)
foreach(header IN LISTS public_headers)
    string(MAKE_C_IDENTIFIER "${header}" identifier)
    set(source "${CMAKE_CURRENT_BINARY_DIR}/header_${identifier}.cpp")
    file(WRITE "${source}" "#include <${header}>\n")
    add_library(header_${identifier} OBJECT "${source}")
    target_link_libraries(header_${identifier} PRIVATE alligator::alligator)
endforeach()
enable_testing()
add_test(NAME alligator_installed_consumer_runs COMMAND alligator_installed_consumer)
set_tests_properties(alligator_installed_consumer_runs PROPERTIES TIMEOUT 30 LABELS packaging)
